# --- core numerics and data handling -----------------------------------------
pandas>=2.2,<3
numpy>=2.2,<3

# --- estimation --------------------------------------------------------------
# statsmodels provides the two-way fixed-effects OLS with cluster-robust standard errors
# used throughout the AI block (12_ai_observable.py). `linearmodels` is deliberately NOT a
# dependency: the panel regressions use explicit C(iso3) + C(year) dummies, not a
# dedicated panel library.
statsmodels>=0.14,<1
scipy>=1.16,<2

# --- machine learning --------------------------------------------------------
# LinearRegression, RandomForestRegressor, HistGradientBoostingRegressor, GroupKFold,
# permutation_importance, and the imputation/scaling pipelines.
scikit-learn>=1.7,<2

# --- plotting ----------------------------------------------------------------
matplotlib>=3.10,<4

# --- data acquisition --------------------------------------------------------
# requests: World Bank v2 API, OpenAlex works API, FAOSTAT bulk ZIPs.
# pycountry: the M49 -> ISO3 and ISO2 -> ISO3 mappings that make the three sources joinable.
requests>=2.32,<3
pycountry>=26.2

# --- manuscript assembly (optional) ------------------------------------------
# Needed only by code/09_build_docx.py, code/09b_build_docx_en.py and
# code/13_build_docx_v2.py, which rebuild the manuscript DOCX. Those scripts read a
# markdown source in article/ that is NOT shipped in this repository, so they will not run
# on a fresh clone. Omit this line if you only want to reproduce the analysis.
python-docx>=1.2,<2
